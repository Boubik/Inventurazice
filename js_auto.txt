(async function () {
  'use strict';

  // ============================================================
  // CONFIGURATION
  // ============================================================
  const CONFIG = {
    // Change this for another building/object.
    buildingPrefix: '64/',

    // Used only when a query is still too large (>9000 records).
    splitChars: '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ-._',

    // Small pause between completed searches.
    delayMs: 300,

    // FIS can take a while for large searches. This is per query.
    queryTimeoutMs: 180000,
    pollMs: 300,

    // Safety limits.
    maxDepth: 7,
    maxRequests: 800,

    roomFieldName: 'mistnost',
    outputFileName: 'evidencni_cisla_auto_64.csv',
    debug: true,
  };

  const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

  function normalizeText(value) {
    return String(value || '')
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')
      .toLowerCase();
  }

  function normalizeRoom(value) {
    return String(value || '')
      .trim()
      .replace(/\\/g, '/')
      .replace(/^\s*(místnost|mistnost|lokalita|umístění|umisteni)\s*:\s*/i, '')
      .trim();
  }

  function locationCode(value) {
    const normalized = normalizeRoom(value);
    if (!normalized) return '';
    return normalized.split(/\s+/)[0];
  }

  function csvCell(value) {
    const text = String(value == null ? '' : value);
    return /[;"\r\n]/.test(text)
      ? '"' + text.replace(/"/g, '""') + '"'
      : text;
  }

  function allSameOriginDocuments(rootDoc) {
    const result = [];
    const seen = new Set();

    function visit(doc) {
      if (!doc || seen.has(doc)) return;
      seen.add(doc);
      result.push(doc);

      for (const frame of doc.querySelectorAll('frame, iframe')) {
        try {
          const child = frame.contentDocument || frame.contentWindow?.document;
          if (child) visit(child);
        } catch (_) {
          // Ignore cross-origin frames.
        }
      }
    }

    visit(rootDoc);
    return result;
  }

  function findWorkingDocument() {
    for (const doc of allSameOriginDocuments(document)) {
      if (doc.querySelector(`[name="${CONFIG.roomFieldName}"]`)) return doc;
    }
    return document;
  }

  function findFilterSubmitter(form) {
    const submits = Array.from(form.querySelectorAll(
      'button[type="submit"], input[type="submit"], input[type="image"]'
    ));

    return submits.find(element => {
      const text = normalizeText(element.value || element.textContent || '');
      return text.includes('filtrovat');
    }) || submits[0] || null;
  }

  function buildFormData(form, roomValue) {
    const data = new FormData(form);
    data.set(CONFIG.roomFieldName, roomValue);

    // Native FormData(form) does not automatically include the clicked submit button.
    const submitter = findFilterSubmitter(form);
    if (submitter && submitter.name) {
      data.append(submitter.name, submitter.value || '');
    }

    return data;
  }

  function toSearchParams(formData) {
    const params = new URLSearchParams();
    for (const [key, value] of formData.entries()) {
      if (typeof value === 'string') params.append(key, value);
    }
    return params;
  }

  function buildGetUrl(form, sourceDoc, roomValue) {
    const base = sourceDoc.location?.href || location.href;
    const url = new URL(form.action || base, base);
    const data = buildFormData(form, roomValue);
    const params = toSearchParams(data);

    // Keep action query parameters but replace parameters also supplied by the form.
    const suppliedNames = new Set();
    for (const [key] of params.entries()) suppliedNames.add(key);
    for (const key of suppliedNames) url.searchParams.delete(key);
    for (const [key, value] of params.entries()) url.searchParams.append(key, value);

    return { url: url.toString(), data };
  }

  function findColumnIndex(table, names) {
    const rows = Array.from(table.querySelectorAll('tr'));
    for (const row of rows) {
      const cells = Array.from(row.querySelectorAll('th, td'));
      if (!cells.length) continue;

      for (let index = 0; index < cells.length; index++) {
        const header = normalizeText(cells[index].innerText || cells[index].textContent || '');
        if (names.some(name => header === name || header.includes(name))) return index;
      }

      // Only the first few rows can reasonably be headers.
      if (rows.indexOf(row) > 4) break;
    }
    return -1;
  }

  function findInventoryTable(doc) {
    const known = doc.querySelector('table.zaznamy');
    if (known) return known;

    for (const table of doc.querySelectorAll('table')) {
      const firstRows = Array.from(table.querySelectorAll('tr')).slice(0, 5);
      const headerText = normalizeText(
        firstRows.map(row => row.innerText || row.textContent || '').join(' | ')
      );

      const hasInventory = headerText.includes('evidencni cislo') || headerText.includes('inventarni cislo');
      const hasName = headerText.includes('nazev');
      const hasLocation = headerText.includes('umisteni') || headerText.includes('mistnost') || headerText.includes('lokalita');

      if (hasInventory && hasName && hasLocation) return table;
    }

    return null;
  }

  function documentText(doc) {
    return normalizeText(doc.body?.innerText || doc.body?.textContent || doc.documentElement?.textContent || '');
  }

  function isTooLarge(doc) {
    const text = documentText(doc);
    return (
      text.includes('vice nez 9000') ||
      text.includes('pozadavek na velky objem dat') ||
      text.includes('velky objem dat')
    );
  }

  function isEmptyResult(doc) {
    const text = documentText(doc);
    return (
      text.includes('nebyly nalezeny zadne zaznamy') ||
      text.includes('nebyl nalezen zadny zaznam') ||
      text.includes('zadne zaznamy') ||
      text.includes('0 zaznamu')
    );
  }

  function isLoadingPage(doc) {
    const text = documentText(doc);
    return (
      text.includes('probiha komunikace se serverem') ||
      text.includes('nacita se vetsi mnozstvi dat')
    );
  }

  function describeDocument(doc) {
    return {
      url: (() => {
        try { return doc.location?.href || ''; } catch (_) { return ''; }
      })(),
      title: doc.title || '',
      tables: doc.querySelectorAll('table').length,
      zaznamy: doc.querySelectorAll('table.zaznamy').length,
      frames: doc.querySelectorAll('frame, iframe').length,
      tooLarge: isTooLarge(doc),
      loading: isLoadingPage(doc),
      empty: isEmptyResult(doc),
      textStart: String(doc.body?.innerText || doc.body?.textContent || '')
        .replace(/\s+/g, ' ')
        .trim()
        .slice(0, 300),
    };
  }

  function extractRowsFromTable(table) {
    const personIndex = findColumnIndex(table, ['odpovedna osoba']);
    const inventoryIndex = findColumnIndex(table, ['evidencni cislo', 'inventarni cislo']);
    const nameIndex = findColumnIndex(table, ['nazev']);
    const locationIndex = findColumnIndex(table, ['umisteni', 'mistnost', 'lokalita']);

    // Current FIS layout from the working manual exporter:
    // Akce=0, Poř.č.=1, Odpovědná osoba=2, Evidenční číslo=3,
    // Název=4, Umístění=5.
    const idxPerson = personIndex >= 0 ? personIndex : 2;
    const idxInventory = inventoryIndex >= 0 ? inventoryIndex : 3;
    const idxName = nameIndex >= 0 ? nameIndex : 4;
    const idxLocation = locationIndex >= 0 ? locationIndex : 5;

    const rows = [];

    for (const tr of table.querySelectorAll('tr')) {
      const cells = Array.from(tr.querySelectorAll('td'));
      if (!cells.length) continue;
      if (Math.max(idxPerson, idxInventory, idxName, idxLocation) >= cells.length) continue;

      const owner = (cells[idxPerson].innerText || cells[idxPerson].textContent || '').trim();
      const inventory = (cells[idxInventory].innerText || cells[idxInventory].textContent || '').trim();
      const name = (cells[idxName].innerText || cells[idxName].textContent || '').trim();
      const room = locationCode(cells[idxLocation].innerText || cells[idxLocation].textContent || '');

      if (!inventory || !name || !room) continue;
      if (!room.startsWith(CONFIG.buildingPrefix)) continue;

      rows.push({ inventory, owner, name, room });
    }

    return rows;
  }

  function inspectLiveFrame(iframe) {
    let rootDoc;
    try {
      rootDoc = iframe.contentDocument || iframe.contentWindow?.document;
    } catch (_) {
      return { state: 'waiting', rows: [], doc: null };
    }

    if (!rootDoc) return { state: 'waiting', rows: [], doc: null };

    const docs = allSameOriginDocuments(rootDoc);

    for (const doc of docs) {
      if (isTooLarge(doc)) return { state: 'too-large', rows: [], doc };
    }

    for (const doc of docs) {
      const table = findInventoryTable(doc);
      if (!table) continue;
      const rows = extractRowsFromTable(table);
      return { state: 'table', rows, doc };
    }

    for (const doc of docs) {
      if (isEmptyResult(doc)) return { state: 'empty', rows: [], doc };
    }

    return {
      state: docs.some(isLoadingPage) ? 'loading' : 'waiting',
      rows: [],
      doc: rootDoc,
    };
  }

  function createWorkerIframe() {
    const old = document.getElementById('inventory-auto-export-worker');
    if (old) old.remove();

    const iframe = document.createElement('iframe');
    iframe.id = 'inventory-auto-export-worker';
    iframe.name = 'inventory-auto-export-worker';

    // Do not use display:none: some old pages behave differently without layout.
    iframe.style.cssText = [
      'position:fixed',
      'left:-10000px',
      'top:0',
      'width:1400px',
      'height:900px',
      'opacity:0.01',
      'pointer-events:none',
      'border:0',
      'z-index:-1',
    ].join(';');

    document.body.appendChild(iframe);
    return iframe;
  }

  async function navigateWorkerGet(iframe, url, roomValue, statusLine) {
    if (CONFIG.debug) console.log(`[auto-export] navigating ${roomValue}`, url);

    // about:blank first prevents us from reading stale results from the previous query.
    iframe.src = 'about:blank';
    await sleep(50);
    iframe.src = url;

    const started = Date.now();
    let lastDebugAt = 0;

    while (Date.now() - started < CONFIG.queryTimeoutMs) {
      const result = inspectLiveFrame(iframe);

      if (result.state === 'too-large' || result.state === 'table' || result.state === 'empty') {
        if (CONFIG.debug && result.doc) {
          console.log(`[auto-export] completed ${roomValue}: ${result.state}`, describeDocument(result.doc));
        }
        return result;
      }

      const elapsed = Math.round((Date.now() - started) / 1000);
      statusLine.textContent = `Hledám: ${roomValue} • čekám ${elapsed} s`;

      if (CONFIG.debug && Date.now() - lastDebugAt > 10000) {
        lastDebugAt = Date.now();
        try {
          const doc = iframe.contentDocument || iframe.contentWindow?.document;
          if (doc) console.log(`[auto-export] waiting ${roomValue}`, describeDocument(doc));
        } catch (_) {
          // Ignore transient navigation state.
        }
      }

      await sleep(CONFIG.pollMs);
    }

    let diagnostic = '';
    try {
      const doc = iframe.contentDocument || iframe.contentWindow?.document;
      if (doc) diagnostic = JSON.stringify(describeDocument(doc));
    } catch (_) {}

    throw new Error(`Timeout while waiting for FIS result for ${roomValue}. ${diagnostic}`);
  }

  function submitWorkerPost(iframe, form, sourceDoc, roomValue) {
    const data = buildFormData(form, roomValue);
    const base = sourceDoc.location?.href || location.href;
    const action = new URL(form.action || base, base).toString();

    const temp = document.createElement('form');
    temp.method = 'POST';
    temp.action = action;
    temp.target = iframe.name;
    temp.style.display = 'none';

    for (const [key, value] of data.entries()) {
      if (typeof value !== 'string') continue;
      const input = document.createElement('input');
      input.type = 'hidden';
      input.name = key;
      input.value = value;
      temp.appendChild(input);
    }

    document.body.appendChild(temp);
    temp.submit();
    temp.remove();
  }

  async function runQuery(iframe, form, sourceDoc, roomValue, statusLine) {
    const method = (form.method || 'GET').toUpperCase();

    if (method === 'GET') {
      const built = buildGetUrl(form, sourceDoc, roomValue);
      if (CONFIG.debug) {
        console.log('[auto-export] query fields', roomValue, Array.from(built.data.entries()));
      }
      return navigateWorkerGet(iframe, built.url, roomValue, statusLine);
    }

    // POST fallback. Current FIS page appears to use GET, but keep this generic.
    iframe.src = 'about:blank';
    await sleep(50);
    submitWorkerPost(iframe, form, sourceDoc, roomValue);

    const started = Date.now();
    while (Date.now() - started < CONFIG.queryTimeoutMs) {
      const result = inspectLiveFrame(iframe);
      if (result.state === 'too-large' || result.state === 'table' || result.state === 'empty') return result;
      const elapsed = Math.round((Date.now() - started) / 1000);
      statusLine.textContent = `Hledám: ${roomValue} • čekám ${elapsed} s`;
      await sleep(CONFIG.pollMs);
    }

    throw new Error(`Timeout while waiting for POST result for ${roomValue}.`);
  }

  function downloadCsv(rows) {
    const building = CONFIG.buildingPrefix.replace(/\/$/, '');
    const data = [[building, `Místnost: více místností objektu ${building}`, '', '']];

    const sorted = Array.from(rows.values()).sort((a, b) => {
      const byRoom = a.room.localeCompare(b.room, 'cs', { numeric: true });
      if (byRoom !== 0) return byRoom;
      return a.inventory.localeCompare(b.inventory, 'cs', { numeric: true });
    });

    for (const row of sorted) data.push([row.inventory, row.owner, row.name, row.room]);

    const csv = data.map(row => row.map(csvCell).join(';')).join('\n');
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = CONFIG.outputFileName;
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  function createStatusBox() {
    const old = document.getElementById('inventory-auto-export-status');
    if (old) old.remove();

    const box = document.createElement('div');
    box.id = 'inventory-auto-export-status';
    box.style.cssText = [
      'position:fixed',
      'right:16px',
      'bottom:16px',
      'z-index:2147483647',
      'width:420px',
      'max-width:calc(100vw - 32px)',
      'padding:14px',
      'background:#1f2327',
      'color:#f5f5f5',
      'border:1px solid #666',
      'border-radius:8px',
      'box-shadow:0 4px 18px rgba(0,0,0,.45)',
      'font:13px/1.4 sans-serif',
    ].join(';');

    box.innerHTML = `
      <div style="font-weight:bold;margin-bottom:6px">Automatický export místností</div>
      <div id="inventory-auto-export-line">Připravuji…</div>
      <div id="inventory-auto-export-stats" style="margin-top:5px;color:#bbb"></div>
      <button id="inventory-auto-export-cancel" style="margin-top:9px;padding:5px 10px">Zrušit</button>
    `;

    document.body.appendChild(box);
    return box;
  }

  // ============================================================
  // MAIN
  // ============================================================
  const sourceDoc = findWorkingDocument();
  const roomControl = sourceDoc.querySelector(`[name="${CONFIG.roomFieldName}"]`);
  if (!roomControl) {
    console.error(`[auto-export] field ${CONFIG.roomFieldName} was not found.`);
    return;
  }

  const form = roomControl.closest('form');
  if (!form) {
    console.error('[auto-export] room filter is not inside a form.');
    return;
  }

  if (CONFIG.debug) {
    console.log('[auto-export] source page', {
      url: sourceDoc.location?.href || location.href,
      roomFieldValue: roomControl.value || '',
      formMethod: form.method || 'GET',
      formAction: form.action || '',
      formTarget: form.target || '',
    });
  }

  const statusBox = createStatusBox();
  const statusLine = statusBox.querySelector('#inventory-auto-export-line');
  const statsLine = statusBox.querySelector('#inventory-auto-export-stats');
  const cancelButton = statusBox.querySelector('#inventory-auto-export-cancel');
  const worker = createWorkerIframe();

  let cancelled = false;
  cancelButton.addEventListener('click', () => {
    cancelled = true;
    try { worker.src = 'about:blank'; } catch (_) {}
    statusLine.textContent = 'Export se ruší…';
  });

  const queue = [{ prefix: CONFIG.buildingPrefix, depth: 0 }];
  const rows = new Map();
  const rooms = new Set();
  const visited = new Set();
  let requests = 0;
  let oversized = 0;
  let empty = 0;

  try {
    while (queue.length && !cancelled) {
      if (requests >= CONFIG.maxRequests) {
        throw new Error(`Safety limit reached: ${CONFIG.maxRequests} requests.`);
      }

      const job = queue.shift();
      if (!job || visited.has(job.prefix)) continue;
      visited.add(job.prefix);

      statusLine.textContent = `Hledám: ${job.prefix}`;
      statsLine.textContent = `Požadavky: ${requests} • fronta: ${queue.length} • místnosti: ${rooms.size} • položky: ${rows.size}`;

      const result = await runQuery(worker, form, sourceDoc, job.prefix, statusLine);
      requests += 1;

      if (result.state === 'too-large') {
        oversized += 1;

        if (job.depth >= CONFIG.maxDepth) {
          throw new Error(`Prefix ${job.prefix} is still too large at maxDepth=${CONFIG.maxDepth}.`);
        }

        for (const char of CONFIG.splitChars) {
          queue.push({ prefix: job.prefix + char, depth: job.depth + 1 });
        }
      } else if (result.state === 'empty') {
        empty += 1;
      } else if (result.state === 'table') {
        if (!result.rows.length) empty += 1;

        for (const row of result.rows) {
          rooms.add(row.room);
          const existing = rows.get(row.inventory);
          if (existing && existing.room !== row.room) {
            console.warn('[auto-export] inventory in multiple locations', row.inventory, existing.room, row.room);
          }
          rows.set(row.inventory, row);
        }
      }

      statsLine.textContent = `Požadavky: ${requests} • fronta: ${queue.length} • místnosti: ${rooms.size} • položky: ${rows.size}`;
      await sleep(CONFIG.delayMs);
    }

    if (cancelled) {
      statusLine.textContent = 'Export zrušen.';
      return;
    }

    if (!rows.size) {
      throw new Error('No inventory rows were found. Send the latest [auto-export] waiting/completed logs from the console.');
    }

    statusLine.textContent = 'Hotovo – vytvářím CSV…';
    statsLine.textContent = `Požadavky: ${requests} • rozdělení: ${oversized} • prázdné: ${empty} • místnosti: ${rooms.size} • položky: ${rows.size}`;

    downloadCsv(rows);
    statusLine.textContent = `Hotovo. Staženo ${rows.size} položek z ${rooms.size} místností.`;

    console.log('[auto-export] finished', {
      requests,
      oversized,
      empty,
      rooms: Array.from(rooms).sort(),
      rows: rows.size,
    });
  } catch (error) {
    console.error('[auto-export] failed', error);
    statusLine.textContent = 'Chyba: ' + (error?.message || String(error));
  } finally {
    // Keep the worker only while debugging a failure; remove it on successful/cancelled run.
    if (!CONFIG.debug || rows.size || cancelled) worker.remove();
  }
})();
