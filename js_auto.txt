(async function () {
  'use strict';

  // ============================================================
  // CONFIGURATION
  // ============================================================
  const CONFIG = {
    // Building/object to export.
    buildingPrefix: '64/',

    // SAFE TEST MODE:
    // Keep this value for the first test. If it works, set it to '' and the
    // script will automatically scan buildingPrefix by smaller prefixes.
    testQuery: '64/5.29',

    // The broad query "64/" is NEVER sent. Automatic mode starts directly
    // with 64/0, 64/1, ..., 64/A, ... to avoid the >9000-record query.
    initialChars: '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ-._',

    // If one of the smaller prefixes is still too large, split it again.
    splitChars: '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz-._ ',

    // Be gentle with IMIS: one request at a time and a pause between them.
    delayMs: 800,
    ajaxTimeoutMs: 120000,

    // Safety limits.
    maxDepth: 7,
    maxRequests: 300,

    roomFieldName: 'mistnost',
    outputFileName: 'evidencni_cisla_auto_64.csv',
    debug: true,
  };

  const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

  function normalizeText(value) {
    return String(value || '')
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')
      .toLowerCase();
  }

  function normalizeRoom(value) {
    return String(value || '')
      .trim()
      .replace(/\\/g, '/')
      .replace(/^\s*(místnost|mistnost|lokalita|umístění|umisteni)\s*:\s*/i, '')
      .trim();
  }

  // IMIS displays values such as:
  //   64/5.29 CPTO Pasteurova/5.29
  // The real location code is only the first token.
  function locationCode(value) {
    const normalized = normalizeRoom(value);
    if (!normalized) return '';
    return normalized.split(/\s+/)[0];
  }

  function csvCell(value) {
    const text = String(value == null ? '' : value);
    return /[;"\r\n]/.test(text)
      ? '"' + text.replace(/"/g, '""') + '"'
      : text;
  }

  function getDocuments(rootDocument) {
    const result = [];
    const seen = new Set();

    function visit(doc) {
      if (!doc || seen.has(doc)) return;
      seen.add(doc);
      result.push(doc);

      for (const frame of doc.querySelectorAll('frame, iframe')) {
        try {
          const child = frame.contentDocument || frame.contentWindow?.document;
          if (child) visit(child);
        } catch (_) {
          // Ignore cross-origin frames.
        }
      }
    }

    visit(rootDocument);
    return result;
  }

  function findImisDocument() {
    for (const doc of getDocuments(document)) {
      if (
        doc.querySelector('#commonFiltersForm') &&
        doc.querySelector('#tabs') &&
        doc.querySelector(`[name="${CONFIG.roomFieldName}"]`)
      ) {
        return doc;
      }
    }

    // When the console is opened in the outer frameset, also inspect top.
    try {
      if (top && top.document && top.document !== document) {
        for (const doc of getDocuments(top.document)) {
          if (
            doc.querySelector('#commonFiltersForm') &&
            doc.querySelector('#tabs') &&
            doc.querySelector(`[name="${CONFIG.roomFieldName}"]`)
          ) {
            return doc;
          }
        }
      }
    } catch (_) {
      // Ignore cross-origin top frame.
    }

    return null;
  }

  function getJQuery(doc) {
    const win = doc.defaultView;
    return (win && (win.jQuery || win.$)) || null;
  }

  function getActiveTabLink(doc, $) {
    let index = -1;

    try {
      index = $('#tabs').tabs('option', 'active');
    } catch (_) {
      // Fall back to DOM state below.
    }

    const links = Array.from(doc.querySelectorAll('#tabs > ul > li > a'));
    if (Number.isInteger(index) && index >= 0 && index < links.length) {
      return links[index];
    }

    return (
      doc.querySelector('#tabs > ul > li.ui-tabs-active > a') ||
      doc.querySelector('#tabs > ul > li[aria-selected="true"] > a') ||
      links[links.length - 1] ||
      null
    );
  }

  function getFilterElements(doc) {
    return Array.from(
      doc.querySelectorAll('#filtersForm .filteritem[name], #commonFiltersForm .filteritem[name]')
    );
  }

  function filterDataForQuery(doc, $, roomValue) {
    // This intentionally mirrors the IMIS code in verso_data/sez_xxBi.js:
    // $('#filtersForm,#commonFiltersForm').find('.filteritem').serialize()
    const data = $('#filtersForm,#commonFiltersForm')
      .find('.filteritem')
      .serializeArray()
      .filter(item => item.name !== CONFIG.roomFieldName);

    data.push({ name: CONFIG.roomFieldName, value: roomValue });
    return data;
  }

  function buildAjaxUrl(doc, $, roomValue) {
    const tabLink = getActiveTabLink(doc, $);
    if (!tabLink) throw new Error('Active IMIS tab link was not found.');

    const base = doc.location?.href || location.href;
    const url = new URL(tabLink.getAttribute('href') || tabLink.href, base);

    // The IMIS tab link can be mutated by previous filtering and contain stale
    // filter parameters. Remove them before appending the current values.
    const filterNames = new Set(getFilterElements(doc).map(element => element.name));
    for (const name of filterNames) url.searchParams.delete(name);

    const data = filterDataForQuery(doc, $, roomValue);
    for (const item of data) url.searchParams.append(item.name, item.value);

    return url.toString();
  }

  function ajaxHtml($, url) {
    return new Promise((resolve, reject) => {
      $.ajax({
        url,
        type: 'GET',
        dataType: 'html',
        cache: false,
        timeout: CONFIG.ajaxTimeoutMs,
      })
        .done(html => resolve(String(html || '')))
        .fail((xhr, status, error) => {
          reject(new Error(`IMIS AJAX failed: ${status}${error ? ' / ' + error : ''} (HTTP ${xhr.status || '?'})`));
        });
    });
  }

  function parseFragment(doc, html) {
    const container = doc.createElement('div');
    container.innerHTML = html;
    return container;
  }

  function fragmentText(container) {
    return normalizeText(container.textContent || '');
  }

  function isTooLarge(container) {
    const text = fragmentText(container);
    return (
      text.includes('vice nez 9000') ||
      text.includes('pozadavek na velky objem dat') ||
      text.includes('velky objem dat')
    );
  }

  function findColumnIndex(table, names) {
    const rows = Array.from(table.querySelectorAll('tr')).slice(0, 5);

    for (const row of rows) {
      const cells = Array.from(row.querySelectorAll('th, td'));
      for (let index = 0; index < cells.length; index++) {
        const text = normalizeText(cells[index].textContent || '');
        if (names.some(name => text === name || text.includes(name))) return index;
      }
    }

    return -1;
  }

  function isInventoryTable(table) {
    const header = normalizeText(
      Array.from(table.querySelectorAll('tr'))
        .slice(0, 5)
        .map(row => row.textContent || '')
        .join(' | ')
    );

    return (
      (header.includes('evidencni cislo') || header.includes('inventarni cislo')) &&
      header.includes('nazev') &&
      (header.includes('umisteni') || header.includes('mistnost') || header.includes('lokalita'))
    );
  }

  function findInventoryTables(container) {
    const known = Array.from(container.querySelectorAll('table.zaznamy'));
    if (known.length) return known;
    return Array.from(container.querySelectorAll('table')).filter(isInventoryTable);
  }

  function extractRowsFromTable(table) {
    const personIndex = findColumnIndex(table, ['odpovedna osoba']);
    const inventoryIndex = findColumnIndex(table, ['evidencni cislo', 'inventarni cislo']);
    const nameIndex = findColumnIndex(table, ['nazev']);
    const locationIndex = findColumnIndex(table, ['umisteni', 'mistnost', 'lokalita']);

    // Fallback indices match the current IMIS table:
    // 0 Akce, 1 Poř.č., 2 Odpovědná osoba, 3 Evidenční číslo,
    // 4 Název, 5 Umístění.
    const idxPerson = personIndex >= 0 ? personIndex : 2;
    const idxInventory = inventoryIndex >= 0 ? inventoryIndex : 3;
    const idxName = nameIndex >= 0 ? nameIndex : 4;
    const idxLocation = locationIndex >= 0 ? locationIndex : 5;

    const rows = [];

    for (const tr of table.querySelectorAll('tr')) {
      const cells = Array.from(tr.querySelectorAll('td'));
      if (!cells.length) continue;
      if (Math.max(idxPerson, idxInventory, idxName, idxLocation) >= cells.length) continue;

      const owner = (cells[idxPerson].textContent || '').trim();
      const inventory = (cells[idxInventory].textContent || '').trim();
      const name = (cells[idxName].textContent || '').trim();
      const room = locationCode(cells[idxLocation].textContent || '');

      if (!inventory || !name || !room) continue;
      if (!room.startsWith(CONFIG.buildingPrefix)) continue;

      rows.push({ inventory, owner, name, room });
    }

    return rows;
  }

  function classifyResponse(doc, html) {
    const container = parseFragment(doc, html);

    if (isTooLarge(container)) {
      return { state: 'too-large', rows: [], container };
    }

    const tables = findInventoryTables(container);
    if (tables.length) {
      const rows = tables.flatMap(extractRowsFromTable);
      return { state: 'table', rows, container };
    }

    const text = fragmentText(container);
    const knownEmpty = (
      text.includes('nebyly nalezeny zadne zaznamy') ||
      text.includes('nebyl nalezen zadny zaznam') ||
      text.includes('zadne zaznamy') ||
      text.includes('0 zaznamu')
    );

    if (knownEmpty || !text.trim()) {
      return { state: 'empty', rows: [], container };
    }

    return { state: 'unknown', rows: [], container };
  }

  async function runQuery(doc, $, roomValue) {
    const url = buildAjaxUrl(doc, $, roomValue);
    if (CONFIG.debug) console.log(`[auto-export] GET ${roomValue}`);

    const html = await ajaxHtml($, url);
    const result = classifyResponse(doc, html);

    if (CONFIG.debug) {
      console.log(`[auto-export] ${roomValue} -> ${result.state}`, {
        rows: result.rows.length,
        responseChars: html.length,
      });
    }

    if (result.state === 'unknown') {
      const preview = String(result.container.textContent || '')
        .replace(/\s+/g, ' ')
        .trim()
        .slice(0, 300);
      throw new Error(`Unexpected IMIS response for ${roomValue}: ${preview}`);
    }

    return result;
  }

  function downloadCsv(rows, suffix = '') {
    const building = CONFIG.buildingPrefix.replace(/\/$/, '');
    const data = [[building, `Místnost: více místností objektu ${building}`, '', '']];

    const sorted = Array.from(rows.values()).sort((a, b) => {
      const byRoom = a.room.localeCompare(b.room, 'cs', { numeric: true });
      if (byRoom !== 0) return byRoom;
      return a.inventory.localeCompare(b.inventory, 'cs', { numeric: true });
    });

    for (const row of sorted) {
      data.push([row.inventory, row.owner, row.name, row.room]);
    }

    const csv = data.map(row => row.map(csvCell).join(';')).join('\n');
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;

    if (suffix) {
      const safeSuffix = suffix.replace(/[^A-Za-z0-9._-]/g, '_');
      link.download = `evidencni_cisla_auto_test_${safeSuffix}.csv`;
    } else {
      link.download = CONFIG.outputFileName;
    }

    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  function createStatusBox(doc) {
    const old = doc.getElementById('inventory-auto-export-status');
    if (old) old.remove();

    const box = doc.createElement('div');
    box.id = 'inventory-auto-export-status';
    box.style.cssText = [
      'position:fixed',
      'right:16px',
      'bottom:16px',
      'z-index:2147483647',
      'width:420px',
      'max-width:calc(100vw - 32px)',
      'padding:14px',
      'background:#1f2327',
      'color:#f5f5f5',
      'border:1px solid #666',
      'border-radius:8px',
      'box-shadow:0 4px 18px rgba(0,0,0,.45)',
      'font:13px/1.4 sans-serif',
    ].join(';');

    box.innerHTML = `
      <div style="font-weight:bold;margin-bottom:6px">Automatický export místností</div>
      <div id="inventory-auto-export-line">Připravuji…</div>
      <div id="inventory-auto-export-stats" style="margin-top:5px;color:#bbb"></div>
      <button id="inventory-auto-export-cancel" style="margin-top:9px;padding:5px 10px">Zrušit</button>
    `;

    doc.body.appendChild(box);
    return box;
  }

  // ============================================================
  // MAIN
  // ============================================================
  const sourceDoc = findImisDocument();
  if (!sourceDoc) {
    console.error('[auto-export] IMIS filter page was not found.');
    return;
  }

  const $ = getJQuery(sourceDoc);
  if (!$ || !$.ajax) {
    console.error('[auto-export] IMIS jQuery was not found.');
    return;
  }

  const statusBox = createStatusBox(sourceDoc);
  const statusLine = statusBox.querySelector('#inventory-auto-export-line');
  const statsLine = statusBox.querySelector('#inventory-auto-export-stats');
  const cancelButton = statusBox.querySelector('#inventory-auto-export-cancel');

  let cancelled = false;
  cancelButton.addEventListener('click', () => {
    cancelled = true;
    statusLine.textContent = 'Export se ruší po dokončení aktuálního požadavku…';
  });

  const rows = new Map();
  const rooms = new Set();
  let requests = 0;
  let oversized = 0;
  let empty = 0;

  function addRows(found) {
    for (const row of found) {
      rooms.add(row.room);
      const existing = rows.get(row.inventory);
      if (existing && existing.room !== row.room) {
        console.warn('[auto-export] inventory in multiple locations', row.inventory, existing.room, row.room);
      }
      rows.set(row.inventory, row);
    }
  }

  try {
    // ----------------------------------------------------------
    // SAFE SINGLE-ROOM TEST
    // ----------------------------------------------------------
    if (CONFIG.testQuery) {
      statusLine.textContent = `Testuji IMIS AJAX: ${CONFIG.testQuery}`;
      statsLine.textContent = 'Proběhne pouze jeden požadavek.';

      const result = await runQuery(sourceDoc, $, CONFIG.testQuery);
      requests = 1;

      if (result.state === 'too-large') {
        throw new Error(`Test query ${CONFIG.testQuery} unexpectedly returned >9000 records.`);
      }

      addRows(result.rows);

      if (!rows.size) {
        throw new Error(`Test query ${CONFIG.testQuery} returned no inventory rows.`);
      }

      downloadCsv(rows, CONFIG.testQuery);
      statusLine.textContent = `TEST OK: ${rows.size} položek, ${rooms.size} místnost(i). CSV staženo.`;
      statsLine.textContent = 'Pokud počet odpovídá ručnímu výsledku, nastavíme testQuery na prázdný řetězec a zapneme automatiku.';

      console.log('[auto-export] SAFE TEST finished', {
        query: CONFIG.testQuery,
        rows: rows.size,
        rooms: Array.from(rooms).sort(),
      });
      return;
    }

    // ----------------------------------------------------------
    // AUTOMATIC DISCOVERY
    // ----------------------------------------------------------
    // Deliberately skip the broad buildingPrefix query itself.
    const queue = Array.from(CONFIG.initialChars).map(char => ({
      prefix: CONFIG.buildingPrefix + char,
      depth: 1,
    }));
    const visited = new Set();

    while (queue.length && !cancelled) {
      if (requests >= CONFIG.maxRequests) {
        throw new Error(`Safety limit reached: ${CONFIG.maxRequests} requests.`);
      }

      const job = queue.shift();
      if (!job || visited.has(job.prefix)) continue;
      visited.add(job.prefix);

      statusLine.textContent = `Hledám: ${job.prefix}`;
      statsLine.textContent = `Požadavky: ${requests} • fronta: ${queue.length} • místnosti: ${rooms.size} • položky: ${rows.size}`;

      const result = await runQuery(sourceDoc, $, job.prefix);
      requests += 1;

      if (result.state === 'too-large') {
        oversized += 1;

        if (job.depth >= CONFIG.maxDepth) {
          throw new Error(`Prefix ${job.prefix} is still too large at maxDepth=${CONFIG.maxDepth}.`);
        }

        for (const char of CONFIG.splitChars) {
          queue.push({ prefix: job.prefix + char, depth: job.depth + 1 });
        }
      } else if (result.state === 'empty') {
        empty += 1;
      } else {
        if (!result.rows.length) empty += 1;
        addRows(result.rows);
      }

      statsLine.textContent = `Požadavky: ${requests} • fronta: ${queue.length} • místnosti: ${rooms.size} • položky: ${rows.size}`;
      await sleep(CONFIG.delayMs);
    }

    if (cancelled) {
      statusLine.textContent = 'Export zrušen.';
      return;
    }

    if (!rows.size) {
      throw new Error('Automatic scan finished without inventory rows.');
    }

    downloadCsv(rows);
    statusLine.textContent = `Hotovo. Staženo ${rows.size} položek z ${rooms.size} místností.`;
    statsLine.textContent = `Požadavky: ${requests} • rozdělení kvůli limitu: ${oversized} • prázdné: ${empty}`;

    console.log('[auto-export] finished', {
      requests,
      oversized,
      empty,
      rooms: Array.from(rooms).sort(),
      rows: rows.size,
    });
  } catch (error) {
    console.error('[auto-export] failed', error);
    statusLine.textContent = 'Chyba: ' + (error?.message || String(error));
  }
})();
