(function () {
  function findDocumentWithTable() {
    if (document.querySelector('table.zaznamy')) return document;
    for (const frame of document.querySelectorAll('frame, iframe')) {
      try {
        const doc = frame.contentDocument || frame.contentWindow.document;
        if (doc && doc.querySelector('table.zaznamy')) return doc;
      } catch (_) {
        // Ignore cross-origin frames.
      }
    }
    return document;
  }

  function valueOf(element) {
    if (!element) return '';
    if (element.tagName === 'SELECT') {
      const option = element.options[element.selectedIndex];
      return (option && (option.value || option.textContent) || '').trim();
    }
    return (element.value || element.textContent || '').trim();
  }

  function escapeRegExp(value) {
    return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  }

  function findFullRoom(doc, rawRoom) {
    const room = (rawRoom || '').trim().replace(/\\/g, '/');
    if (!room || room.includes('/')) return room;

    const bodyText = doc.body ? doc.body.innerText : '';
    const pattern = new RegExp('(?:^|[^A-Za-z0-9._/-])([A-Za-z0-9._-]+/' + escapeRegExp(room) + ')(?=$|[^A-Za-z0-9._/-])', 'gm');
    const matches = new Set();
    let match;
    while ((match = pattern.exec(bodyText)) !== null) matches.add(match[1]);
    if (matches.size === 1) return Array.from(matches)[0];

    const selectors = [
      '[name="budova"]', '[name="objekt"]', '[name="lokalita"]',
      '[id*="budova" i]', '[id*="objekt" i]', '[id*="lokalita" i]'
    ];
    for (const selector of selectors) {
      const candidate = valueOf(doc.querySelector(selector)).replace(/\\/g, '/').trim();
      if (!candidate) continue;
      if (candidate.includes('/') && candidate.endsWith('/' + room)) return candidate;
      if (/^[A-Za-z0-9._-]+$/.test(candidate) && candidate !== room) return candidate + '/' + room;
    }

    console.warn('Could not determine a building prefix for room:', room);
    return room;
  }

  function csvCell(value) {
    const text = String(value == null ? '' : value);
    if (/[;"\r\n]/.test(text)) return '"' + text.replace(/"/g, '""') + '"';
    return text;
  }

  const doc = findDocumentWithTable();
  const table = doc.querySelector('table.zaznamy');
  if (!table) {
    console.error('Table with class "zaznamy" not found.');
    return;
  }

  const roomControl = doc.querySelector('input[name="mistnost"], select[name="mistnost"], [name="mistnost"]');
  const rawRoom = valueOf(roomControl);
  const roomValue = findFullRoom(doc, rawRoom);
  const data = [];

  if (roomValue) data.push([roomValue, `Místnost: ${roomValue}`, '']);

  table.querySelectorAll('tr').forEach(row => {
    const cells = row.querySelectorAll('td');
    if (cells.length > 4) {
      const osoba = cells[2].innerText.trim();
      const evidencni = cells[3].innerText.trim();
      const nazev = cells[4].innerText.trim();
      if (evidencni && osoba && nazev) data.push([evidencni, osoba, nazev]);
    }
  });

  if (!data.length) {
    console.log('No data entries found.');
    return;
  }

  const csvContent = data.map(row => row.map(csvCell).join(';')).join('\n');
  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const fileName = `evidencni_cisla_${roomValue || 'vse'}.csv`.replace(/[^a-zA-Z0-9._-]/g, '_');

  if (navigator.msSaveBlob) {
    navigator.msSaveBlob(blob, fileName);
  } else {
    const link = document.createElement('a');
    link.href = URL.createObjectURL(blob);
    link.download = fileName;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(link.href);
  }

  console.log(`Exported ${data.length - (roomValue ? 1 : 0)} inventory rows for room ${roomValue || '(unknown)'}.`);
})();
